AULA_7: Configurando Patch

    Pop-up de aviso da Movimentação:
        Botão Cadastro;
            Formula: Patch(Movimentação; Default(Movimentação);
                        {ID_produto: Gallery.Selected.Id;
                        Qauntidade: Value(txt_quantidade.text);
                        Tipo_movimentação: txt_tipo.text}
                     );

                     Patch(Produtos; LookUp(Produtos;ID = Gallery.Selected.Id);
                        {Qauntidade: 
                            If(txt_tipo.text = 'Entrada'; 
                                Value(txt_quantidade.text) + Gallery.Selected.Qauntidade_estoque; 
                                Value(txt_quantidade.text) - Gallery.Selected.Qauntidade_estoque
                            )}
                     );

                     Notify("Movimentação cadastrada com sucesso!")
